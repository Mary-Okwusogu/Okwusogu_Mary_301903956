Baseline Query 
=======================
SELECT approval_id, transaction_id, approver_customer_id, mandate_id, decision, decided_at
FROM transaction_approval
WHERE DATE(decided_at) BETWEEN '2026-02-05' AND '2026-02-18'
ORDER BY decided_at; 


Optimised Query
=====================
SELECT approval_id, transaction_id, approver_customer_id, mandate_id, decision, decided_at
FROM transaction_approval
WHERE decided_at >= '2026-02-05 00:00:00'
  AND decided_at <  '2026-02-19 00:00:00'
ORDER BY decided_at;
